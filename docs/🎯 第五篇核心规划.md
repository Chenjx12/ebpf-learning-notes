### 1. eBPF 中的函数调用（BPF-to-BPF）

#### 1.1 函数调用基础

**示例代码 (`simple-func.c`)：**

```
// simple-func.c
#include <uapi/linux/ptrace.h>
#include <linux/sched.h>

// 定义事件结构体
struct data_t {
    u32 pid;
    char comm[16];
};

BPF_PERF_OUTPUT(events);

// 辅助函数：获取进程信息
static __always_inline void get_proc_info(struct data_t *data) {
    data->pid = bpf_get_current_pid_tgid() >> 32;
    bpf_get_current_comm(&data->comm, sizeof(data->comm));
}

// 主函数
TRACEPOINT_PROBE(syscalls, sys_enter_execve) {
    struct data_t data = {};
    
    // 调用辅助函数
    get_proc_info(&data);
    
    events.perf_submit(args, &data, sizeof(data));
    return 0;
}
```

#### 1.2 函数调用的限制

**⚠️ 重要限制：**

1. **栈大小限制**：eBPF 只有 512 字节栈，函数调用会叠加栈帧
2. **调用深度限制**：目前内核最多支持 8 层函数调用
3. **性能开销**：每次函数调用都有额外的压栈/出栈开销

**实验：观察栈消耗**

```
# 编译并运行 simple-func.py
sudo python3 simple-func.py

# 在另一个终端观察栈使用情况
sudo bpftool prog dump xlated id <ID> | grep "r10"
```

### 2. 尾调用（Tail Call）：不回来的调用

#### 2.1 尾调用原理

**关键概念：**

- **普通调用**：调用后返回，栈帧叠加
- **尾调用**：调用后不返回，复用当前栈帧

**尾调用流程：**

调用调用不返回A程序A程序B程序C结束直接退出

#### 2.2 尾调用实现：`prog_array` Map

**核心机制：**

```
// 定义 prog_array
struct {
    __uint(type, BPF_MAP_TYPE_PROG_ARRAY);
    __uint(max_entries, 8);  // 最多8个程序
    __type(key, __u32);      // 索引
    __type(value, __u32);    // 程序文件描述符
} jmp_table SEC(".maps");

// 尾调用
bpf_tail_call(ctx, &jmp_table, index);
```

#### 2.3 尾调用示例 (`hello-tail.c`)

```
// hello-tail.c
#include <uapi/linux/ptrace.h>
#include <linux/bpf.h>

// 定义 prog_array
struct {
    __uint(type, BPF_MAP_TYPE_PROG_ARRAY);
    __uint(max_entries, 4);
    __type(key, __u32);
    __type(value, __u32);
} jmp_table SEC(".maps");

// 许可证
char LICENSE[] SEC("license") = "GPL";

// 程序1：入口
SEC("kprobe/sys_execve")
int hello_entry(struct pt_regs *ctx) {
    bpf_trace_printk("Hello from ENTRY program!");
    // 尾调用到程序2
    bpf_tail_call(ctx, &jmp_table, 1);
    // 如果尾调用失败，继续执行
    bpf_trace_printk("Fallback in ENTRY program");
    return 0;
}

// 程序2：处理
SEC("kprobe/sys_execve")
int hello_handler(struct pt_regs *ctx) {
    bpf_trace_printk("Hello from HANDLER program!");
    // 可以继续尾调用到下一个程序
    // bpf_tail_call(ctx, &jmp_table, 2);
    return 0;
}
```

### 3. 实验：构建程序链

#### 3.1 Python 加载器 (`hello-tail.py`)

```
#!/usr/bin/python3
from bcc import BPF

# 加载eBPF程序
b = BPF(src_file="hello-tail.c")

# 获取函数对象
entry_fn = b.load_func("hello_entry", BPF.KPROBE)
handler_fn = b.load_func("hello_handler", BPF.KPROBE)

# 获取jmp_table
jmp_table = b.get_table("jmp_table")

# 设置程序链：entry -> handler
jmp_table[0] = b.ctypes.c_uint32(entry_fn.fd)  # 索引0 -> entry
jmp_table[1] = b.ctypes.c_uint32(handler_fn.fd) # 索引1 -> handler

# 附加entry程序到kprobe
b.attach_kprobe(event=b.get_syscall_fnname("execve"), fn_name="hello_entry")

print("Tracing execve with tail calls... Hit Ctrl-C to end.")
b.trace_print()
```
